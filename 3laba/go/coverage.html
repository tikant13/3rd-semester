import (

// Array — динамический массив строк (как Array в C++).
// Поля с маленькой буквы закрыты (private), методы с большой — открыты (public).
type Array struct {
        data []string // выделенная память, len(data) — ёмкость
        size int      // сколько элементов записано
}

func NewArray() *Array {
        return &Array{data: make([]string, 10)}
}

// grow увеличивает ёмкость в 2 раза, когда массив заполнен.
func (a *Array) grow() {
        newCapacity := len(a.data) * 2
        if newCapacity == 0 {
                newCapacity = 10
        }
        newData := make([]string, newCapacity)
        copy(newData, a.data[:a.size])
        a.data = newData
}

func (a *Array) PushBack(value string) {
        if a.size == len(a.data) {
                a.grow()
        }
        a.data[a.size] = value
        a.size++
}

// Insert вставляет по индексу; index == Size() — вставка в конец.
func (a *Array) Insert(index int, value string) bool {
        if index < 0 || index > a.size {
                return false
        }
        if a.size == len(a.data) {
                a.grow()
        }
        for i := a.size; i > index; i-- {
                a.data[i] = a.data[i-1]
        }
        a.data[index] = value
        a.size++
        return true
}

func (a *Array) Get(index int) (string, bool) {
        if index < 0 || index >= a.size {
                return "", false
        }
        return a.data[index], true
}

func (a *Array) Set(index int, value string) bool {
        if index < 0 || index >= a.size {
                return false
        }
        a.data[index] = value
        return true
}

func (a *Array) Delete(index int) bool {
        if index < 0 || index >= a.size {
                return false
        }
        for i := index; i < a.size-1; i++ {
                a.data[i] = a.data[i+1]
        }
        a.size--
        a.data[a.size] = ""
        return true
}

func (a *Array) Size() int {
        return a.size
}

// String — все элементы через пробел.
func (a *Array) String() string {
        return strings.Join(a.data[:a.size], " ")
}

// ---------- Сериализация ----------
// Load* читает во временный объект и только при успехе заменяет текущий,
// поэтому при ошибке массив не меняется.

func (a *Array) SaveText(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                for i := 0; i < a.size; i++ {
                        writeLine(w, a.data[i])
                }
        })
}

func (a *Array) LoadText(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                tmp := NewArray()
                for _, word := range readWords(r) {
                        tmp.PushBack(word)
                }
                *a = *tmp
                return nil
        })
}

func (a *Array) SaveBinary(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                writeInt(w, a.size)
                for i := 0; i < a.size; i++ {
                        writeString(w, a.data[i])
                }
        })
}

func (a *Array) LoadBinary(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                count, err := readCount(r)
                if err != nil {
                        return err
                }
                tmp := NewArray()
                for i := 0; i < count; i++ {
                        value, err := readString(r)
                        if err != nil {
                                return err
                        }
                        tmp.PushBack(value)
                }
                *a = *tmp
                return nil
        })
}

// ChainTable — хеш-таблица методом цепочек (как ChainTable в C++):
// массив корзин, каждая корзина — односвязный список с записями "ключ\tзначение".
type ChainTable struct {
        table []*LinkedList // корзины
        count int           // количество пар
}

func NewChainTable(capacity int) *ChainTable {
        if capacity < 1 {
                capacity = 1
        }
        t := &ChainTable{table: make([]*LinkedList, capacity)}
        for i := range t.table {
                t.table[i] = NewLinkedList()
        }
        return t
}

// hash: h = h*31 + символ, затем остаток от деления на число корзин.
func (t *ChainTable) hash(key string) int {
        var h uint64
        for i := 0; i < len(key); i++ {
                h = h*31 + uint64(key[i])
        }
        return int(h % uint64(len(t.table)))
}

// findInBucket ищет запись с ключом key в корзине b; возвращает её или "".
func (t *ChainTable) findInBucket(b int, key string) (string, bool) {
        bucket := t.table[b]
        for i := 0; i < bucket.Size(); i++ {
                entry, _ := bucket.Get(i)
                if k, _, _ := strings.Cut(entry, "\t"); k == key {
                        return entry, true
                }
        }
        return "", false
}

// Insert: true — новый ключ, false — значение обновлено (или ключ пустой).
func (t *ChainTable) Insert(key, value string) bool {
        if key == "" {
                return false
        }
        b := t.hash(key)
        if entry, found := t.findInBucket(b, key); found {
                t.table[b].DelValue(entry)
                t.table[b].PushFront(key + "\t" + value)
                return false
        }
        t.table[b].PushFront(key + "\t" + value)
        t.count++
        if t.count*4 > len(t.table)*3 { // заполнено больше 75%
                t.rehash()
        }
        return true
}

func (t *ChainTable) Get(key string) (string, bool) {
        entry, found := t.findInBucket(t.hash(key), key)
        if !found {
                return "", false
        }
        _, value, _ := strings.Cut(entry, "\t")
        return value, true
}

func (t *ChainTable) Remove(key string) bool {
        b := t.hash(key)
        entry, found := t.findInBucket(b, key)
        if !found {
                return false
        }
        t.table[b].DelValue(entry)
        t.count--
        return true
}

// rehash увеличивает число корзин в 2 раза и раскладывает записи заново.
func (t *ChainTable) rehash() {
        old := t.table
        t.table = make([]*LinkedList, len(old)*2)
        for i := range t.table {
                t.table[i] = NewLinkedList()
        }
        for _, bucket := range old {
                for entry, ok := bucket.Head(); ok; entry, ok = bucket.Head() {
                        key, _, _ := strings.Cut(entry, "\t")
                        t.table[t.hash(key)].PushFront(entry)
                        bucket.DelFront()
                }
        }
}

func (t *ChainTable) Size() int {
        return t.count
}

func (t *ChainTable) Capacity() int {
        return len(t.table)
}

// forEach вызывает visit для каждой пары.
func (t *ChainTable) forEach(visit func(key, value string)) {
        for _, bucket := range t.table {
                for i := 0; i < bucket.Size(); i++ {
                        entry, _ := bucket.Get(i)
                        key, value, _ := strings.Cut(entry, "\t")
                        visit(key, value)
                }
        }
}

// String — пары "ключ значение" через пробел.
func (t *ChainTable) String() string {
        var parts []string
        t.forEach(func(key, value string) {
                parts = append(parts, key+" "+value)
        })
        return strings.Join(parts, " ")
}

// ---------- Сериализация ----------
// Сохраняются только пары; при чтении они вставляются заново,
// поэтому раскладка по корзинам может стать другой.

func (t *ChainTable) SaveText(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                t.forEach(func(key, value string) {
                        writeLine(w, key, value)
                })
        })
}

func (t *ChainTable) LoadText(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                words := readWords(r)
                if len(words)%2 != 0 {
                        return errBadFile // у ключа нет значения
                }
                tmp := NewChainTable(8)
                for i := 0; i < len(words); i += 2 {
                        tmp.Insert(words[i], words[i+1])
                }
                *t = *tmp
                return nil
        })
}

func (t *ChainTable) SaveBinary(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                writeInt(w, t.count)
                t.forEach(func(key, value string) {
                        writeString(w, key)
                        writeString(w, value)
                })
        })
}

func (t *ChainTable) LoadBinary(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                count, err := readCount(r)
                if err != nil {
                        return err
                }
                tmp := NewChainTable(8)
                for i := 0; i < count; i++ {
                        key, err := readString(r)
                        if err != nil {
                                return err
                        }
                        value, err := readString(r)
                        if err != nil {
                                return err
                        }
                        tmp.Insert(key, value)
                }
                *t = *tmp
                return nil
        })
}

import (

        "strings"

// maxKicks — сколько раз подряд можно выталкивать, потом считаем, что попали в цикл.
const maxKicks = 100

// CuckooTable — хеширование кукушки (как CuckooTable в C++): две таблицы и две
// хеш-функции. Ключ лежит либо в T1[h1(key)], либо в T2[h2(key)].
// Таблицы — массивы Array, пустая ячейка — пустая строка.
type CuckooTable struct {
        keys1, vals1 *Array // первая таблица
        keys2, vals2 *Array // вторая таблица
        capacity     int    // размер каждой таблицы
        count        int    // количество пар
}

func NewCuckooTable(capacity int) *CuckooTable {
        if capacity < 1 {
                capacity = 1
        }
        t := &CuckooTable{}
        t.allocate(capacity)
        return t
}

// allocate создаёт две пустые таблицы размера capacity.
func (t *CuckooTable) allocate(capacity int) {
        t.capacity = capacity
        t.keys1, t.vals1, t.keys2, t.vals2 = NewArray(), NewArray(), NewArray(), NewArray()
        for i := 0; i < capacity; i++ {
                t.keys1.PushBack("")
                t.vals1.PushBack("")
                t.keys2.PushBack("")
                t.vals2.PushBack("")
        }
}

// hash: полиномиальный хеш с основанием base.
func (t *CuckooTable) hash(key string, base uint64) int {
        var h uint64
        for i := 0; i < len(key); i++ {
                h = h*base + uint64(key[i])
        }
        return int(h % uint64(t.capacity))
}

func (t *CuckooTable) h1(key string) int { return t.hash(key, 31) }
func (t *CuckooTable) h2(key string) int { return t.hash(key, 37) }

// cell возвращает ключ и значение ячейки i таблицы (keys, vals).
func cell(keys, vals *Array, i int) (string, string) {
        k, _ := keys.Get(i)
        v, _ := vals.Get(i)
        return k, v
}

// keyAt возвращает ключ ячейки i ("" — ячейка пустая).
func keyAt(keys *Array, i int) string {
        k, _ := keys.Get(i)
        return k
}

// place кладёт пару в T1; если занято — выталкивает старую пару в T2 и так далее.
// При неудаче возвращает false и пару, оставшуюся без места.
func (t *CuckooTable) place(key, value string) (string, string, bool) {
        for step := 0; step < maxKicks; step++ {
                i := t.h1(key)
                oldKey, oldValue := cell(t.keys1, t.vals1, i)
                t.keys1.Set(i, key)
                t.vals1.Set(i, value)
                if oldKey == "" {
                        return "", "", true
                }
                key, value = oldKey, oldValue

                j := t.h2(key)
                oldKey, oldValue = cell(t.keys2, t.vals2, j)
                t.keys2.Set(j, key)
                t.vals2.Set(j, value)
                if oldKey == "" {
                        return "", "", true
                }
                key, value = oldKey, oldValue
        }
        return key, value, false
}

// rehash собирает все пары (и лишнюю extra, если есть) и раскладывает
// в таблицы размера newCapacity; если не влезло — увеличивает ещё в 2 раза.
func (t *CuckooTable) rehash(newCapacity int, extraKey, extraValue string) {
        var keys, values []string
        t.forEach(func(key, value string) {
                keys = append(keys, key)
                values = append(values, value)
        })
        if extraKey != "" {
                keys = append(keys, extraKey)
                values = append(values, extraValue)
        }
        for capacity := newCapacity; ; capacity *= 2 {
                t.allocate(capacity)
                ok := true
                for i := range keys {
                        if _, _, placed := t.place(keys[i], values[i]); !placed {
                                ok = false
                                break
                        }
                }
                if ok {
                        return
                }
        }
}

// Insert: true — новый ключ, false — значение обновлено (или ключ пустой).
func (t *CuckooTable) Insert(key, value string) bool {
        if key == "" {
                return false // пустая строка обозначает пустую ячейку
        }
        if i := t.h1(key); keyAt(t.keys1, i) == key { // ключ уже есть — меняем значение
                t.vals1.Set(i, value)
                return false
        }
        if j := t.h2(key); keyAt(t.keys2, j) == key {
                t.vals2.Set(j, value)
                return false
        }
        // при заполнении больше 45% кукушка часто зацикливается — увеличиваем заранее
        if (t.count+1)*100 > t.capacity*2*45 {
                t.rehash(t.capacity*2, "", "")
        }
        if k, v, placed := t.place(key, value); !placed {
                t.rehash(t.capacity*2, k, v) // перестраиваем вместе с парой без места
        }
        t.count++
        return true
}

func (t *CuckooTable) Get(key string) (string, bool) {
        if key == "" {
                return "", false
        }
        if k, v := cell(t.keys1, t.vals1, t.h1(key)); k == key {
                return v, true
        }
        if k, v := cell(t.keys2, t.vals2, t.h2(key)); k == key {
                return v, true
        }
        return "", false
}

func (t *CuckooTable) Remove(key string) bool {
        if key == "" {
                return false
        }
        if i := t.h1(key); keyAt(t.keys1, i) == key {
                t.keys1.Set(i, "")
                t.vals1.Set(i, "")
                t.count--
                return true
        }
        if j := t.h2(key); keyAt(t.keys2, j) == key {
                t.keys2.Set(j, "")
                t.vals2.Set(j, "")
                t.count--
                return true
        }
        return false
}

func (t *CuckooTable) Size() int {
        return t.count
}

// forEach вызывает visit для каждой занятой ячейки T1 и T2.
func (t *CuckooTable) forEach(visit func(key, value string)) {
        for i := 0; i < t.capacity; i++ {
                if k, v := cell(t.keys1, t.vals1, i); k != "" {
                        visit(k, v)
                }
                if k, v := cell(t.keys2, t.vals2, i); k != "" {
                        visit(k, v)
                }
        }
}

// String — пары "ключ значение" через пробел.
func (t *CuckooTable) String() string {
        var parts []string
        t.forEach(func(key, value string) {
                parts = append(parts, key+" "+value)
        })
        return strings.Join(parts, " ")
}

// ---------- Сериализация ----------
// Сохраняются только пары; при чтении они вставляются заново,
// поэтому ячейки T1/T2 могут стать другими.

func (t *CuckooTable) SaveText(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                t.forEach(func(key, value string) {
                        writeLine(w, key, value)
                })
        })
}

func (t *CuckooTable) LoadText(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                words := readWords(r)
                if len(words)%2 != 0 {
                        return errBadFile // у ключа нет значения
                }
                tmp := NewCuckooTable(8)
                for i := 0; i < len(words); i += 2 {
                        tmp.Insert(words[i], words[i+1])
                }
                *t = *tmp
                return nil
        })
}

func (t *CuckooTable) SaveBinary(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                writeInt(w, t.count)
                t.forEach(func(key, value string) {
                        writeString(w, key)
                        writeString(w, value)
                })
        })
}

        return loadFile(fileName, func(r *bufio.Reader) error {
                count, err := readCount(r)
                if err != nil {
                        return err
                }
                tmp := NewCuckooTable(8)
                for i := 0; i < count; i++ {
                        key, err := readString(r)
                        if err != nil {
                                return err
                        }
                        value, err := readString(r)
                        if err != nil {
                                return err
                        }

                }
                *t = *tmp
                return nil

// doublyNode — узел двусвязного списка (закрытый тип).
type doublyNode struct {
        value string
        next  *doublyNode
        prev  *doublyNode
}

// DoublyList — двусвязный список (как DoublyList в C++).
type DoublyList struct {
        head *doublyNode
        tail *doublyNode
}

func NewDoublyList() *DoublyList {
        return &DoublyList{}
}

func (l *DoublyList) PushFront(value string) {
        node := &doublyNode{value: value, next: l.head}
        if l.head == nil {
                l.tail = node
        } else {
                l.head.prev = node
        }
        l.head = node
}

func (l *DoublyList) PushBack(value string) {
        node := &doublyNode{value: value, prev: l.tail}
        if l.tail == nil {
                l.head = node
        } else {
                l.tail.next = node
        }
        l.tail = node
}

// find возвращает первый узел со значением value или nil.
func (l *DoublyList) find(value string) *doublyNode {
        for cur := l.head; cur != nil; cur = cur.next {
                if cur.value == value {
                        return cur
                }
        }
        return nil
}

// remove вырезает узел из списка, поправляя соседей.
func (l *DoublyList) remove(node *doublyNode) {
        if node.prev == nil {
                l.head = node.next
        } else {
                node.prev.next = node.next
        }
        if node.next == nil {
                l.tail = node.prev
        } else {
                node.next.prev = node.prev
        }
}

func (l *DoublyList) PushBefore(target, value string) bool {
        node := l.find(target)
        if node == nil {
                return false
        }
        if node == l.head {
                l.PushFront(value)
                return true
        }
        newNode := &doublyNode{value: value, prev: node.prev, next: node}
        node.prev.next = newNode
        node.prev = newNode
        return true
}

func (l *DoublyList) PushAfter(target, value string) bool {
        node := l.find(target)
        if node == nil {
                return false
        }
        if node == l.tail {
                l.PushBack(value)
                return true
        }
        newNode := &doublyNode{value: value, prev: node, next: node.next}
        node.next.prev = newNode
        node.next = newNode
        return true
}

func (l *DoublyList) Head() (string, bool) {
        if l.head == nil {
                return "", false
        }
        return l.head.value, true
}

func (l *DoublyList) Tail() (string, bool) {
        if l.tail == nil {
                return "", false
        }
        return l.tail.value, true
}

func (l *DoublyList) Get(index int) (string, bool) {
        if index < 0 {
                return "", false
        }
        cur := l.head
        for i := 0; i < index && cur != nil; i++ {
                cur = cur.next
        }
        if cur == nil {
                return "", false
        }
        return cur.value, true
}

func (l *DoublyList) DelFront() bool {
        if l.head == nil {
                return false
        }
        l.remove(l.head)
        return true
}

func (l *DoublyList) DelBack() bool {
        if l.tail == nil {
                return false
        }
        l.remove(l.tail)
        return true
}

func (l *DoublyList) DelBefore(target string) bool {
        node := l.find(target)
        if node == nil || node.prev == nil {
                return false
        }
        l.remove(node.prev)
        return true
}

func (l *DoublyList) DelAfter(target string) bool {
        node := l.find(target)
        if node == nil || node.next == nil {
                return false
        }
        l.remove(node.next)
        return true
}

func (l *DoublyList) DelValue(value string) bool {
        node := l.find(value)
        if node == nil {
                return false
        }
        l.remove(node)
        return true
}

func (l *DoublyList) Find(value string) int {
        index := 0
        for cur := l.head; cur != nil; cur = cur.next {
                if cur.value == value {
                        return index
                }
                index++
        }
        return -1
}

func (l *DoublyList) Size() int {
        count := 0
        for cur := l.head; cur != nil; cur = cur.next {
                count++
        }
        return count
}

func (l *DoublyList) String() string {
        var parts []string
        for cur := l.head; cur != nil; cur = cur.next {
                parts = append(parts, cur.value)
        }
        return strings.Join(parts, " ")
}

// Backward — элементы от хвоста к голове (проверка связей prev).
func (l *DoublyList) Backward() string {
        var parts []string
        for cur := l.tail; cur != nil; cur = cur.prev {
                parts = append(parts, cur.value)
        }
        return strings.Join(parts, " ")
}

// ---------- Сериализация (от головы к хвосту) ----------

func (l *DoublyList) SaveText(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                for cur := l.head; cur != nil; cur = cur.next {
                        writeLine(w, cur.value)
                }
        })
}

func (l *DoublyList) LoadText(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                tmp := NewDoublyList()
                for _, word := range readWords(r) {
                        tmp.PushBack(word)
                }
                *l = *tmp
                return nil
        })
}

func (l *DoublyList) SaveBinary(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                writeInt(w, l.Size())
                for cur := l.head; cur != nil; cur = cur.next {
                        writeString(w, cur.value)
                }
        })
}

func (l *DoublyList) LoadBinary(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                count, err := readCount(r)
                if err != nil {
                        return err
                }
                tmp := NewDoublyList()
                for i := 0; i < count; i++ {
                        value, err := readString(r)
                        if err != nil {
                                return err
                        }
                        tmp.PushBack(value)
                }
                *l = *tmp
                return nil
        })
}

package ds

import (
        "bufio"
        "encoding/binary"
        "errors"
        "fmt"
        "io"
        "os"
)

// Общие функции для сериализации. Сами методы SaveText/LoadText/SaveBinary/LoadBinary
// находятся у каждой структуры. Функции с маленькой буквы не видны вне пакета (private).

// errBadFile — файл открылся, но его содержимое не подходит.
var errBadFile = errors.New("файл повреждён")

// saveFile создаёт файл и передаёт буфер функции write.
// bufio.Writer запоминает первую ошибку записи, Flush её возвращает.
func saveFile(fileName string, write func(w *bufio.Writer)) error {
        file, err := os.Create(fileName)
        if err != nil {
                return err
        }
        w := bufio.NewWriter(file)
        write(w)
        if err := w.Flush(); err != nil {
                _ = file.Close()
                return err
        }
        return file.Close()
}

// loadFile открывает файл и передаёт его функции read.
func loadFile(fileName string, read func(r *bufio.Reader) error) error {
        file, err := os.Open(fileName)
        if err != nil {
                return err
        }
        err = read(bufio.NewReader(file))
        _ = file.Close() // файл только читали, ошибка закрытия не важна
        return err
}

// writeLine пишет значения через пробел и перевод строки (для текстового формата).
func writeLine(w io.Writer, values ...any) {
        _, _ = fmt.Fprintln(w, values...)
}

// readWords читает все слова текстового файла (разделители — пробелы и переводы строк).
func readWords(r io.Reader) []string {
        scanner := bufio.NewScanner(r)
        scanner.Split(bufio.ScanWords)
        var words []string
        for scanner.Scan() {
                words = append(words, scanner.Text())
        }
        return words
}

// В бинарном файле число занимает 4 байта (int32, младший байт первым),
// как int в C++-версии на x86-64.
func writeInt(w io.Writer, value int) {
        _ = binary.Write(w, binary.LittleEndian, int32(value))
}

func readInt(r io.Reader) (int, error) {
        var value int32
        err := binary.Read(r, binary.LittleEndian, &value)
        return int(value), err
}

// readCount читает количество элементов: отрицательное значит, что файл повреждён.
func readCount(r io.Reader) (int, error) {
        count, err := readInt(r)
        if err != nil {
                return 0, err
        }
        if count < 0 {
                return 0, errBadFile
        }
        return count, nil
}

// Строка записывается как длина, затем её байты.
func writeString(w io.Writer, value string) {
        writeInt(w, len(value))
        _, _ = io.WriteString(w, value)
}

func readString(r io.Reader) (string, error) {
        length, err := readCount(r)
        if err != nil {
                return "", err
        }
        buf := make([]byte, length)
        if _, err := io.ReadFull(r, buf); err != nil {
                return "", err
        }
        return string(buf), nil
}

import (

        "strings"

// listNode — узел односвязного списка (закрытый тип).
type listNode struct {
        value string
        next  *listNode
}

// LinkedList — односвязный список (как linkedList в C++).
type LinkedList struct {
        head *listNode
        tail *listNode
}

func NewLinkedList() *LinkedList {
        return &LinkedList{}
}

func (l *LinkedList) PushFront(value string) {
        l.head = &listNode{value: value, next: l.head}
        if l.tail == nil {
                l.tail = l.head
        }
}

func (l *LinkedList) PushBack(value string) {
        node := &listNode{value: value}
        if l.tail == nil {
                l.head = node
        } else {
                l.tail.next = node
        }
        l.tail = node
}

// findPrev возвращает узел перед первым узлом со значением target.
// found == false — такого значения нет; prev == nil — target в голове.
func (l *LinkedList) findPrev(target string) (prev *listNode, found bool) {
        for cur := l.head; cur != nil; cur = cur.next {
                if cur.value == target {
                        return prev, true
                }
                prev = cur
        }
        return nil, false
}

// PushBefore добавляет value перед первым target.
func (l *LinkedList) PushBefore(target, value string) bool {
        prev, found := l.findPrev(target)
        if !found {
                return false
        }
        if prev == nil {
                l.PushFront(value)
                return true
        }
        prev.next = &listNode{value: value, next: prev.next}
        return true
}

// PushAfter добавляет value после первого target.
func (l *LinkedList) PushAfter(target, value string) bool {
        prev, found := l.findPrev(target)
        if !found {
                return false
        }
        node := l.head
        if prev != nil {
                node = prev.next
        }
        if node == l.tail {
                l.PushBack(value)
                return true
        }
        node.next = &listNode{value: value, next: node.next}
        return true
}

func (l *LinkedList) Head() (string, bool) {
        if l.head == nil {
                return "", false
        }
        return l.head.value, true
}

func (l *LinkedList) Tail() (string, bool) {
        if l.tail == nil {
                return "", false
        }
        return l.tail.value, true
}

func (l *LinkedList) Get(index int) (string, bool) {
        if index < 0 {
                return "", false
        }
        cur := l.head
        for i := 0; i < index && cur != nil; i++ {
                cur = cur.next
        }
        if cur == nil {
                return "", false
        }
        return cur.value, true
}

func (l *LinkedList) DelFront() bool {
        if l.head == nil {
                return false
        }
        l.head = l.head.next
        if l.head == nil {
                l.tail = nil
        }
        return true
}

func (l *LinkedList) DelBack() bool {
        if l.head == nil {
                return false
        }
        if l.head == l.tail {
                l.head, l.tail = nil, nil
                return true
        }
        cur := l.head
        for cur.next != l.tail {
                cur = cur.next
        }
        cur.next = nil
        l.tail = cur
        return true
}

// removeAfter удаляет узел после prev (prev == nil — удалить голову).
func (l *LinkedList) removeAfter(prev *listNode) {
        if prev == nil {
                l.DelFront()
                return
        }
        del := prev.next
        prev.next = del.next
        if del == l.tail {
                l.tail = prev
        }
}

// DelValue удаляет первый узел со значением value.
func (l *LinkedList) DelValue(value string) bool {
        prev, found := l.findPrev(value)
        if !found {
                return false
        }
        l.removeAfter(prev)
        return true
}

// DelBefore удаляет элемент перед первым target.
func (l *LinkedList) DelBefore(target string) bool {
        var prevPrev, prev *listNode // два узла перед текущим
        for cur := l.head; cur != nil; cur = cur.next {
                if cur.value == target {
                        if prev == nil {
                                return false // перед головой ничего нет
                        }
                        l.removeAfter(prevPrev) // удаляем prev
                        return true
                }
                prevPrev, prev = prev, cur
        }
        return false
}

// DelAfter удаляет элемент после первого target.
func (l *LinkedList) DelAfter(target string) bool {
        prev, found := l.findPrev(target)
        if !found {
                return false
        }
        node := l.head
        if prev != nil {
                node = prev.next
        }
        if node.next == nil {
                return false // после хвоста ничего нет
        }
        l.removeAfter(node)
        return true
}

// Find возвращает индекс первого value или -1.
func (l *LinkedList) Find(value string) int {
        index := 0
        for cur := l.head; cur != nil; cur = cur.next {
                if cur.value == value {
                        return index
                }
                index++
        }
        return -1
}

func (l *LinkedList) Size() int {
        count := 0
        for cur := l.head; cur != nil; cur = cur.next {
                count++
        }
        return count
}

func (l *LinkedList) String() string {
        var parts []string
        for cur := l.head; cur != nil; cur = cur.next {
                parts = append(parts, cur.value)
        }
        return strings.Join(parts, " ")
}

// ---------- Сериализация (от головы к хвосту) ----------

func (l *LinkedList) SaveText(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                for cur := l.head; cur != nil; cur = cur.next {
                        writeLine(w, cur.value)
                }
        })
}

func (l *LinkedList) LoadText(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                tmp := NewLinkedList()
                for _, word := range readWords(r) {
                        tmp.PushBack(word)
                }
                *l = *tmp
                return nil
        })
}

func (l *LinkedList) SaveBinary(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                writeInt(w, l.Size())
                for cur := l.head; cur != nil; cur = cur.next {
                        writeString(w, cur.value)
                }
        })
}

func (l *LinkedList) LoadBinary(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                count, err := readCount(r)
                if err != nil {
                        return err
                }
                tmp := NewLinkedList()
                for i := 0; i < count; i++ {
                        value, err := readString(r)
                        if err != nil {
                                return err
                        }
                        tmp.PushBack(value)
                }
                *l = *tmp
                return nil
        })
}

package ds

import (
        "bufio"
        "strconv"
        "strings"
)

const (
        red   = 'R'
        black = 'B'
)

// treeNode — узел красно-чёрного дерева (закрытый тип).
type treeNode struct {
        key    int
        color  byte
        left   *treeNode
        right  *treeNode
        parent *treeNode
}

// RBTree — красно-чёрное дерево целых ключей (как RBTree в C++).
type RBTree struct {
        root *treeNode
}

func NewRBTree() *RBTree {
        return &RBTree{}
}

// isRed: пустой узел (nil) считается чёрным.
func isRed(node *treeNode) bool {
        return node != nil && node.color == red
}

// Левый поворот вокруг x: правый ребёнок y поднимается на место x.
func (t *RBTree) leftRotate(x *treeNode) {
        y := x.right
        x.right = y.left
        if y.left != nil {
                y.left.parent = x
        }
        t.replaceChild(x, y)
        y.left = x
        x.parent = y
}

func (t *RBTree) rightRotate(x *treeNode) {
        y := x.left
        x.left = y.right
        if y.right != nil {
                y.right.parent = x
        }
        t.replaceChild(x, y)
        y.right = x
        x.parent = y
}

// replaceChild ставит v на место u у родителя u (transplant).
func (t *RBTree) replaceChild(u, v *treeNode) {
        if u.parent == nil {
                t.root = v
        } else if u == u.parent.left {
                u.parent.left = v
        } else {
                u.parent.right = v
        }
        if v != nil {
                v.parent = u.parent
        }
}

// Insert добавляет ключ; false — такой ключ уже есть.
func (t *RBTree) Insert(key int) bool {
        var parent *treeNode
        cur := t.root
        for cur != nil {
                parent = cur
                if key == cur.key {
                        return false
                }
                if key < cur.key {
                        cur = cur.left
                } else {
                        cur = cur.right
                }
        }
        node := &treeNode{key: key, color: red, parent: parent} // новый узел всегда красный
        if parent == nil {
                t.root = node
        } else if key < parent.key {
                parent.left = node
        } else {
                parent.right = node
        }
        t.insertFixup(node)
        return true
}

// insertFixup убирает два красных узла подряд.
func (t *RBTree) insertFixup(node *treeNode) {
        for node != t.root && node.parent.color == red {
                parent := node.parent
                grand := parent.parent
                if parent == grand.left {
                        uncle := grand.right
                        if isRed(uncle) { // дядя красный: перекрашиваем и поднимаемся
                                parent.color, uncle.color, grand.color = black, black, red
                                node = grand
                                continue
                        }
                        if node == parent.right {
                                node = parent
                                t.leftRotate(node)
                                parent = node.parent
                        }
                        parent.color, grand.color = black, red
                        t.rightRotate(grand)
                } else {
                        uncle := grand.left
                        if isRed(uncle) {
                                parent.color, uncle.color, grand.color = black, black, red
                                node = grand
                                continue
                        }
                        if node == parent.left {
                                node = parent
                                t.rightRotate(node)
                                parent = node.parent
                        }
                        parent.color, grand.color = black, red
                        t.leftRotate(grand)
                }
        }
        t.root.color = black
}

func (t *RBTree) findNode(key int) *treeNode {
        cur := t.root
        for cur != nil && cur.key != key {
                if key < cur.key {
                        cur = cur.left
                } else {
                        cur = cur.right
                }
        }
        return cur
}

func (t *RBTree) Contains(key int) bool {
        return t.findNode(key) != nil
}

// Remove удаляет ключ с сохранением свойств дерева; false — ключа нет.
func (t *RBTree) Remove(key int) bool {
        z := t.findNode(key)
        if z == nil {
                return false
        }
        y := z // узел, который уходит со своего места
        yColor := y.color
        var x, xParent *treeNode // x встаёт на место y; x может быть nil, поэтому храним родителя
        switch {
        case z.left == nil:
                x, xParent = z.right, z.parent
                t.replaceChild(z, z.right)
        case z.right == nil:
                x, xParent = z.left, z.parent
                t.replaceChild(z, z.left)
        default:
                y = z.right // минимальный в правом поддереве
                for y.left != nil {
                        y = y.left
                }
                yColor = y.color
                x = y.right
                if y.parent == z {
                        xParent = y
                } else {
                        xParent = y.parent
                        t.replaceChild(y, y.right)
                        y.right = z.right
                        y.right.parent = y
                }
                t.replaceChild(z, y)
                y.left = z.left
                y.left.parent = y
                y.color = z.color
        }
        if yColor == black { // ушёл чёрный узел — восстанавливаем чёрную высоту
                t.deleteFixup(x, xParent)
        }
        return true
}

func (t *RBTree) deleteFixup(x, xParent *treeNode) {
        for x != t.root && !isRed(x) {
                if x == xParent.left {
                        brother := xParent.right
                        if isRed(brother) {
                                brother.color, xParent.color = black, red
                                t.leftRotate(xParent)
                                brother = xParent.right
                        }
                        if !isRed(brother.left) && !isRed(brother.right) {
                                brother.color = red
                                x, xParent = xParent, xParent.parent
                                continue
                        }
                        if !isRed(brother.right) {
                                brother.left.color, brother.color = black, red
                                t.rightRotate(brother)
                                brother = xParent.right
                        }
                        brother.color, xParent.color = xParent.color, black
                        brother.right.color = black
                        t.leftRotate(xParent)
                        x = t.root
                } else { // то же самое зеркально
                        brother := xParent.left
                        if isRed(brother) {
                                brother.color, xParent.color = black, red
                                t.rightRotate(xParent)
                                brother = xParent.left
                        }
                        if !isRed(brother.left) && !isRed(brother.right) {
                                brother.color = red
                                x, xParent = xParent, xParent.parent
                                continue
                        }
                        if !isRed(brother.left) {
                                brother.right.color, brother.color = black, red
                                t.leftRotate(brother)
                                brother = xParent.left
                        }
                        brother.color, xParent.color = xParent.color, black
                        brother.left.color = black
                        t.rightRotate(xParent)
                        x = t.root
                }
        }
        if x != nil {
                x.color = black
        }
}

// Обходы возвращают ключи в нужном порядке.

func (t *RBTree) Inorder() []int {
        var keys []int
        var walk func(node *treeNode)
        walk = func(node *treeNode) {
                if node != nil {
                        walk(node.left)
                        keys = append(keys, node.key)
                        walk(node.right)
                }
        }
        walk(t.root)
        return keys
}

func (t *RBTree) Preorder() []int {
        var keys []int
        t.forEachPreorder(func(node *treeNode) {
                keys = append(keys, node.key)
        })
        return keys
}

func (t *RBTree) Postorder() []int {
        var keys []int
        var walk func(node *treeNode)
        walk = func(node *treeNode) {
                if node != nil {
                        walk(node.left)
                        walk(node.right)
                        keys = append(keys, node.key)
                }
        }
        walk(t.root)
        return keys
}

// BFS — обход в ширину (по уровням).
func (t *RBTree) BFS() []int {
        var keys []int
        if t.root == nil {
                return keys
        }
        queue := []*treeNode{t.root}
        for len(queue) > 0 {
                node := queue[0]
                queue = queue[1:]
                keys = append(keys, node.key)
                if node.left != nil {
                        queue = append(queue, node.left)
                }
                if node.right != nil {
                        queue = append(queue, node.right)
                }
        }
        return keys
}

func (t *RBTree) Size() int {
        return len(t.Preorder())
}

// String — пары "ключ цвет" в прямом обходе (как ToString в C++).
func (t *RBTree) String() string {
        var parts []string
        t.forEachPreorder(func(node *treeNode) {
                parts = append(parts, strconv.Itoa(node.key)+" "+string(node.color))
        })
        return strings.Join(parts, " ")
}

func (t *RBTree) forEachPreorder(visit func(node *treeNode)) {
        var walk func(node *treeNode)
        walk = func(node *treeNode) {
                if node != nil {
                        visit(node)
                        walk(node.left)
                        walk(node.right)
                }
        }
        walk(t.root)
}

// load ставит узел с готовым цветом на место по правилу BST (для чтения файла).
// Если вставлять узлы в прямом обходе, получится то же самое дерево.
func (t *RBTree) load(key int, color byte) {
        var parent *treeNode
        cur := t.root
        for cur != nil {
                parent = cur
                if key < cur.key {
                        cur = cur.left
                } else {
                        cur = cur.right
                }
        }
        node := &treeNode{key: key, color: color, parent: parent}
        if parent == nil {
                t.root = node
        } else if key < parent.key {
                parent.left = node
        } else {
                parent.right = node
        }
}

// ---------- Сериализация (прямой обход: ключ и цвет) ----------

func (t *RBTree) SaveText(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                t.forEachPreorder(func(node *treeNode) {
                        writeLine(w, node.key, string(node.color))
                })
        })
}

func (t *RBTree) LoadText(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                words := readWords(r)
                if len(words)%2 != 0 {
                        return errBadFile // у последнего ключа нет цвета
                }
                tmp := NewRBTree()
                for i := 0; i < len(words); i += 2 {
                        key, err := strconv.Atoi(words[i])
                        if err != nil {
                                return errBadFile // ключ не число
                        }
                        color := words[i+1]
                        if color != "R" && color != "B" {
                                return errBadFile
                        }
                        tmp.load(key, color[0])
                }
                *t = *tmp
                return nil
        })
}

func (t *RBTree) SaveBinary(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                writeInt(w, t.Size())
                t.forEachPreorder(func(node *treeNode) {
                        writeInt(w, node.key)
                        _ = w.WriteByte(node.color)
                })
        })
}

func (t *RBTree) LoadBinary(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                count, err := readCount(r)
                if err != nil {
                        return err
                }
                tmp := NewRBTree()
                for i := 0; i < count; i++ {
                        key, err := readInt(r)
                        if err != nil {
                                return err
                        }
                        color, err := r.ReadByte()
                        if err != nil {
                                return err
                        }
                        if color != red && color != black {
                                return errBadFile
                        }
                        tmp.load(key, color)
                }
                *t = *tmp
                return nil
        })
}

package ds

import (
        "bufio"
        "slices"
        "strings"
)

// Stack — стек на односвязном списке (как Stack в C++).
type Stack struct {
        top *listNode // вершина
}

func NewStack() *Stack {
        return &Stack{}
}

func (s *Stack) Push(value string) {
        s.top = &listNode{value: value, next: s.top}
}

// Pop снимает вершину; false — стек пуст.
func (s *Stack) Pop() (string, bool) {
        if s.top == nil {
                return "", false
        }
        value := s.top.value
        s.top = s.top.next
        return value, true
}

func (s *Stack) Peek() (string, bool) {
        if s.top == nil {
                return "", false
        }
        return s.top.value, true
}

func (s *Stack) Size() int {
        count := 0
        for cur := s.top; cur != nil; cur = cur.next {
                count++
        }
        return count
}

// String — элементы от дна к вершине (как ToString в C++).
func (s *Stack) String() string {
        var parts []string
        for cur := s.top; cur != nil; cur = cur.next {
                parts = append(parts, cur.value)
        }
        slices.Reverse(parts)
        return strings.Join(parts, " ")
}

// ---------- Сериализация (от вершины ко дну) ----------
// При чтении элементы идут от вершины ко дну, поэтому Push делаем с конца списка.

func (s *Stack) SaveText(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                for cur := s.top; cur != nil; cur = cur.next {
                        writeLine(w, cur.value)
                }
        })
}

func (s *Stack) LoadText(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                words := readWords(r)
                tmp := NewStack()
                for i := len(words) - 1; i >= 0; i-- {
                        tmp.Push(words[i])
                }
                *s = *tmp
                return nil
        })
}

func (s *Stack) SaveBinary(fileName string) error {
        return saveFile(fileName, func(w *bufio.Writer) {
                writeInt(w, s.Size())
                for cur := s.top; cur != nil; cur = cur.next {
                        writeString(w, cur.value)
                }
        })
}

func (s *Stack) LoadBinary(fileName string) error {
        return loadFile(fileName, func(r *bufio.Reader) error {
                count, err := readCount(r)
                if err != nil {
                        return err
                }
                var values []string
                for i := 0; i < count; i++ {
                        value, err := readString(r)
                        if err != nil {
                                return err
                        }
                        values = append(values, value)
                }
                tmp := NewStack()
                for i := len(values) - 1; i >= 0; i-- {
                        tmp.Push(values[i])
                }
                *s = *tmp
                return nil
        })
}
